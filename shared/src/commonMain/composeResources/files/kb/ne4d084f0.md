}  
  
// Задача

 * Но LoadUrlUseCase должен кэшировать ответы * и ограничивать количество одновременных запросов * список ответов должен быть в том же порядке что и список запросов */

// Data  
interface GetCachedData {  
    suspend fun getCachedData(url: String): Data?  
    suspend fun addAndReturn(url: String, data: Data): Data?  
}  
  
interface RateLimiter<T> {  
    suspend fun withRateLimit(block: suspend () -> T): T  
}  
  
class RateLimiterImpl<T>(  
    limit: Int,  
) : RateLimiter<T> {  
    private val semaphore = Semaphore(limit)  
    override suspend fun withRateLimit(block: suspend () -> T): T = semaphore.withPermit { block() }  
}  
  
interface Cache<K, V> {  
    suspend fun withCache(key: K, block: suspend () -> V): V  
}  
  
class CacheImpl<K, V> : Cache<K, V> {  
    private val cache = mutableMapOf<K, V>()  
    private val locks = mutableMapOf<K, Mutex>()  
    private val locksMutex = Mutex()  
  
    override suspend fun withCache(key: K, block: suspend () -> V): V {  
        return cache[key] ?: getOrPutMutex(key)  
            .withLock { cache.getOrPut(key) { block() } }  
            .also { removeMutex(key) }  
    }  
  
    private suspend fun getOrPutMutex(key: K) = locksMutex.withLock { locks.getOrPut(key) { Mutex() } }  
    private suspend fun removeMutex(key: K) = locksMutex.withLock { locks.remove(key) }  
  
}  
  
interface LoadUrlRepository {  
    suspend fun load(url: URL): Data  
}  
// Data.Repository  
class LoadUrlRepositoryImpl(  
    private val loadUrlDataSource: LoadUrlDataSource,  
    private val cache: Cache<URL, Data>,  
    private val rateLimiter: RateLimiter<Data>,  
) : LoadUrlRepository {  
    override suspend fun load(url: URL): Data {  
        return cache.withCache(key = url) {  
            rateLimiter.withRateLimit { loadUrlDataSource.load(url) }  
        }    }  
  
}  
  
class LoadUrlUseCaseImpl(  
    val repository: LoadUrlRepository,  
) : LoadUrlUseCase {  
  
    // Структура данных для ответов. HashMap, где ключ URL -> Data  
    // Ограничение количества одновременных    // Проходимся по списку урлов, потом делаем async ждем deffered и вызываем awaitAll() и ждем Data  
    override suspend fun load(urls: List<URL>): List<Data> = coroutineScope {  
        urls.map {  
            async { repository.load(it) }  
        }.awaitAll()  
    }  
}

## Решение 2 (проще): `Map<URL, Deferred<Data>>` + `Semaphore`

Вся конструкция `Cache` + `RateLimiter` + `Repository` + per-key мьютексы заменяется одним классом. Идея: кэшируем не результат, а **`Deferred<Data>`** — тогда «замком» служит сам `async`: первый пришедший запускает загрузку, все остальные (в том числе из других вызовов `load`) получают тот же `Deferred` и просто ждут его. Дедупликация in-flight запросов получается бесплатно, без единого ручного `Mutex`.

```kotlin
import kotlinx.coroutines.*
import kotlinx.coroutines.sync.Semaphore
import kotlinx.coroutines.sync.withPermit
import java.util.concurrent.ConcurrentHashMap

class LoadUrlUseCaseImpl(
    private val dataSource: LoadUrlDataSource,
    limit: Int = 4,
) : LoadUrlUseCase {

    // Свой scope: закэшированный Deferred должен жить дольше одного вызова load()
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)

    private val semaphore = Semaphore(limit)
    private val cache = ConcurrentHashMap<URL, Deferred<Data>>()

    override suspend fun load(urls: List<URL>): List<Data> =
        urls.map { url ->
            cache.computeIfAbsent(url) {
                scope.async {
                    semaphore.withPermit { dataSource.load(url) }
                }
            }
        }.awaitAll()
}

fun main(): Unit = runBlocking {
    val useCase = LoadUrlUseCaseImpl(LoadUrlDataSourceImpl(), limit = 2)

    val urls = listOf(URL("a"), URL("b"), URL("a"), URL("c"))

    println(useCase.load(urls))   // ~2 сек: "a" грузится один раз, лимит = 2
    println(useCase.load(urls))   // мгновенно: всё из кэша
}
```

Как закрыты требования:

- **Кэш** — `ConcurrentHashMap`, потокобезопасна сама по себе. `computeIfAbsent` атомарен: два одновременных промаха по одному URL создадут ровно один `async`. Внутри лямбды `computeIfAbsent` нельзя суспендиться — но `async` только *создаёт* корутину, не ждёт её, поэтому всё легально.
- **Лимит одновременных запросов** — корутинный `Semaphore(limit)`: внутри `dataSource.load` одновременно не больше `limit` корутин, остальные подвисают на suspend. Пермит берётся уже внутри `async`, т.е. только на реальную загрузку — попадания в кэш семафор не трогают.
- **Порядок ответов = порядок запросов** — `urls.map { ... }.awaitAll()`: `awaitAll` возвращает результаты в порядке списка, а не по мере готовности.
- **Дедупликация** — и внутри одного списка (`URL("a")` дважды), и между конкурентными вызовами `load`: все ждут один и тот же `Deferred`.

Что назвать слабостью самому (и как дожать, если спросят):

1. **Ошибка кэшируется навсегда**: упавший `Deferred` останется в мапе и будет перебрасывать исключение всем следующим. Лечится очисткой по завершении с ошибкой:
   ```kotlin
   cache.computeIfAbsent(url) {
       scope.async { semaphore.withPermit { dataSource.load(url) } }
           .also { d -> d.invokeOnCompletion { e -> if (e != null) cache.remove(url, d) } }
   }
   ```
2. **Свой `scope` = свой жизненный цикл**: нужен `SupervisorJob` (иначе одна упавшая загрузка отменит соседние) и метод `close()` / привязка к жизненному циклу владельца. Запускать в `coroutineScope` вызова нельзя — отмена первого клиента убила бы `Deferred` для всех остальных.
3. Кэш ничем не ограничен — в проде нужен TTL/размер (`Caffeine`), но для интервью достаточно проговорить.

> [!info] Что делает `computeIfAbsent`
> «Дай значение по ключу, а если его нет — вычисли, положи и верни». Псевдокод:
> ```kotlin
> val existing = map[key]
> if (existing != null) return existing
> val newValue = lambda(key)   // вызывается ТОЛЬКО если ключа нет
> map[key] = newValue
> return newValue
> ```
> **Главное — атомарность.** На `ConcurrentHashMap` вся цепочка «проверить → вычислить → записать» выполняется как одна неделимая операция: на время вычисления блокируется бакет с этим ключом. Две корутины одновременно промахнулись по одному URL → лямбда выполнится **ровно один раз**: первая создаст `async`, вторая подождёт на замке бакета и получит уже готовый `Deferred`. Именно это заменяет всю ручную конструкцию `locks: Map<K, Mutex>` + `locksMutex`.
>
> **Отличие от `getOrPut`:** это extension Kotlin из двух раздельных вызовов (get, потом put) — без синхронизации. Две корутины могут обе не найти ключ, обе вычислить значение, одно перезапишет другое. Ловушка: `getOrPut` **не атомарен даже на `ConcurrentHashMap`** — для атомарности нужен именно `computeIfAbsent`.
>
> **Ограничения (красиво звучат на собеседовании):**
> - Лямбда выполняется под замком бакета → должна быть **короткой и неблокирующей**. У нас соблюдено: `scope.async { }` только *создаёт* корутину и сразу возвращает `Deferred`, сама загрузка идёт уже вне замка. Суспендиться внутри и не выйдет — лямбда не `suspend`.
> - Внутри лямбды **нельзя модифицировать ту же мапу** (другие ключи) — документированное ограничение `ConcurrentHashMap`, иначе deadlock/`IllegalStateException`.
> - Вернёт `null` → запись не добавляется (для `async` неактуально).
>
> Формула для интервью: «`computeIfAbsent` — атомарный get-or-create; фабрика вызовется один раз на ключ, поэтому `async` внутри него даёт бесплатную дедупликацию конкурентных запросов».

> [!abstract] Три требования задачи и чем они закрыты
> 1. **Кэширование** → `Cache` (`getOrPut` под per-key `Mutex`).
> 2. **Лимит одновременных запросов** → `RateLimiter` на `Semaphore(limit)` + `withPermit`.
> 3. **Порядок ответов = порядок запросов** → `urls.map { async { … } }.awaitAll()`: `map` сохраняет порядок, `awaitAll` возвращает результаты в том же порядке (а не по мере готовности).

> [!tip] Зачем per-key Mutex, а не один общий
> Ключевая идея — **дедупликация одновременных запросов одного URL**. Без per-key замка два параллельных `async` для одинакового URL оба не найдут значение в кэше и сделают **два сетевых вызова**. Per-key `Mutex` гарантирует, что грузит только первый, а остальные дождутся и возьмут готовое из кэша. `locksMutex` защищает саму мапу замков (`getOrPut`/`remove`) от гонки.

> [!warning] Тонкое место: `.also { removeMutex(key) }`
> Удаление замка сразу после загрузки экономит память, но при высокой конкуренции возможен момент, когда один корутин уже удалил `Mutex` для ключа, а другой в этот момент создал **новый** через `getOrPut` — и два корутина окажутся на разных замках. Фастпас `cache[key] ?: …` это смягчает (после первой записи в кэш блокировка не нужна), но строго корректнее либо **не удалять** замки, либо хранить значение как `Deferred<V>` в кэше (тогда дедупликация — на уровне самого `async`, без ручных замков):
> ```kotlin
> private val inFlight = mutableMapOf<K, Deferred<V>>()
> // под locksMutex: inFlight.getOrPut(key) { scope.async { block() } }
> ```

> [!note] `value class` как ключ кэша
> `URL`/`Data` — `@JvmInline value class`. У них автогенерируются `equals`/`hashCode` по обёрнутому значению, поэтому `URL` корректно работает ключом `Map`. При этом на рантайме чаще всего нет аллокации обёртки — плюс к производительности.

## Как отвечать на собеседовании

> [!tip] Главный инсайт задачи
> Здесь проверяют **декомпозицию требований на примитивы конкурентности**, а не архитектуру. Три требования → три инструмента: кэш → потокобезопасная мапа, лимит одновременных запросов → `Semaphore` + `withPermit`, порядок ответов → `map { async }` + `awaitAll()`. Решение выше (Repository + RateLimiter-интерфейс + мапа per-key мьютексов) — **переусложнено для интервью**: начинать надо с 15 строк, а усложнять только когда дожмут.

> [!example] Скрипт ответа (порядок важен)
> 1. **Проговорить требования вслух и инструмент под каждое:** «кэш — `ConcurrentHashMap`, лимит — `Semaphore(n)`, порядок — `awaitAll()` возвращает в порядке создания, а не готовности».
> 2. **Написать минимальный вариант** — ноль мьютексов:
> ```kotlin
> class LoadUrlUseCaseImpl(
>     private val dataSource: LoadUrlDataSource,
>     limit: Int = 4,
> ) : LoadUrlUseCase {
>     private val cache = ConcurrentHashMap<URL, Data>()
>     private val semaphore = Semaphore(limit)
>
>     override suspend fun load(urls: List<URL>): List<Data> = coroutineScope {
>         urls.map { url ->
>             async {
>                 cache[url] ?: semaphore.withPermit { dataSource.load(url) }
>                     .also { cache[url] = it }
>             }
>         }.awaitAll()
>     }
> }
> ```
> 3. **Самому назвать слабость** (не ждать вопроса): два одновременных промаха по одному URL → два сетевых вызова. Не ошибка (второй перезапишет тем же значением), но лишний запрос.
> 4. **Если дожимают про дубли в одном списке** → `urls.distinct()` перед загрузкой, потом собрать `Map<URL, Data>` и восстановить порядок через `urls.map { loaded.getValue(it) }`.
> 5. **Если дожимают про дедупликацию между вызовами** → кэшировать `Deferred<Data>` вместо `Data`: один `Mutex` только вокруг `getOrPut`, «замком» служит сам `async` — все ждут один и тот же `Deferred`. Это замена всей конструкции `locks: Map<K, Mutex>` + `locksMutex` + `removeMutex`.

> [!check] Чек-лист ловушек в задачах «кэш + конкурентность»
> - **`Mutex` вокруг `ConcurrentHashMap` не нужен** — мапа уже потокобезопасна. Двойная защита — минус, а не плюс.
> - **`Semaphore` из `kotlinx.coroutines.sync`**, не `java.util.concurrent` — корутинный подвешивает (suspend), джавовый блокирует поток.
> - **`awaitAll()` сохраняет порядок** создания `Deferred` — это и закрывает требование «порядок ответов = порядок запросов», отдельно сортировать не надо.
> - **Кэшируешь `Deferred` → запускай в внешнем `scope`**, не в `coroutineScope` вызова: иначе отмена/ошибка первого клиента убьёт закэшированный `Deferred` для всех остальных. И при исключении — удалять запись, чтобы не закэшировать ошибку навсегда.
> - **Удаление per-key мьютекса (`removeMutex`) — гонка:** один удалил, другой в этот момент создал новый через `getOrPut` → два корутина на разных замках. Если уж per-key замки — не удалять их, либо перейти на `Deferred`-кэш.
> - **Семафор брать только на промахе кэша** (фастпас `cache[url] ?: ...`) — иначе попадания в кэш тоже съедают пермиты и лимитируют друг друга.

> [!abstract] Как себя вести
> - **Начинать с простого и работающего**, усложнять по запросу. «Вот минимум, закрывающий все три пункта; вот что улучшу, если нужна дедупликация» — выглядит сильнее, чем сразу трёхэтажная архитектура.
> - **Называть слабости своего решения до того, как их найдёт интервьюер** — это главный сигнал сеньорности в таких задачах.
> - **Терминами:** «дедупликация in-flight запросов», «фастпас без блокировки», «suspend vs blocking семафор», «structured concurrency и владение scope».
> - Упомянуть продакшен-альтернативы одной строкой: в реальном коде это `Caffeine`/`LoadingCache` или `Flow` + `shareIn`, но на интервью пишем руками.
