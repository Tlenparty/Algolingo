import java.util.concurrent.Executors  
  
class Counter(var count:Long){  
    fun increment(){  
        count++  
    }  
  
    fun decrement(){  
        count--  
    }  
}  
  
val counter = Counter(0)  
  
fun main() {  
  
    val run1 = Runnable {  
        for (i in 0..99){  
            counter.increment()  
        }  
    }  
  
    val run2 = Runnable {  
        for (i in 0..99){  
            counter.decrement()  
        }  
    }  
  
    val executor = Executors.newCachedThreadPool()  
    executor.execute(run1)  
    executor.execute(run2)  
    println(counter.count)  
}
