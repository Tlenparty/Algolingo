open class Formula1DataSource private constructor(  
    var context: Context,  
    var service: ApiService,  
) {  
  
    fun loadAllData(): LinkedList<Any> {  
        val engines: LinkedList<Engine> = service.loadAvailableEngines()  
        val result = LinkedList<Any>()  
  
        for (i in 0..engines.size) {  
            if (engines[i].volume >= Const.TRaceVolume) {  
                result.add(engines[i])  
            }  
            if (engines[i].volume >= TRaceCylinders) {  
                result.add(engines[i])  
            }  
        }  
  
        if(result.size == 0){  
            throw Exception("No engines available")  
        }  
        return result  
    }  
  
    @Synchronized  
    fun getLastCar(): FormulaCar? = car  
  
    @Synchronized  
    fun saveCar(engine: Engine, name: String) {  
        if (car == null) {  
            car = FormulaCar(engine, context.getString(R.string.app_name), "Tinkoff")  
        } else {  
            car?.engine = engine  
            car?.name = context.getString(R.string.app_name)  
        }  
    }  
  
    companion object {  
        var car: FormulaCar? = null  
        fun getInstance(context: Context, service: ApiService) = Formula1DataSource(context.applicationContext, service)  
    }

Реализация репозитория добавления и обращения к списку с помощью корутин

```kotlin

import kotlinx.coroutines.*
import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock

data class Car(val name: String, val id: Int)

class CarRepository {
    private val cars = mutableListOf<Car>()
    private val mutex = Mutex() // Для синхронизации доступа

    // Добавление нового автомобиля (потокобезопасно)
    suspend fun addCar(car: Car) {
        mutex.withLock {
            cars.add(car)
        }
    }

    // Получение последнего добавленного автомобиля (потокобезопасно)
    suspend fun getLastCar(): Car? {
        return mutex.withLock {
            cars.lastOrNull() // Берем последний элемент списка
        }
    }
}

// Тест многопоточности
fun main() = runBlocking {
    val repository = CarRepository()

    // Запуск корутин для добавления машин
    val addJob = launch(Dispatchers.Default) {
        repeat(5) { i ->
            repository.addCar(Car("Car_$i", i))
            println("Добавлена машина: Car_$i")
            delay(500) // Эмуляция задержки
        }
    }

    // Запуск корутин для получения последнего добавленного объекта
    val getJob = launch(Dispatchers.IO) {
        repeat(5) {
            val lastCar = repository.getLastCar()
            println("Последний добавленный: $lastCar")
            delay(700) // Чуть больше задержка, чтобы происходила конкуренция потоков
        }
    }

    addJob.join() // Ждем завершения добавления
    getJob.join() // Ждем завершения получения
}

```

> [!danger] Баги в исходном коде, которые надо назвать на ревью
> 1. **`for (i in 0..engines.size)`** — диапазон `0..size` **включает** `size` → `IndexOutOfBoundsException`. Нужно `0 until engines.size` или, лучше, итерация по элементам/`filter`.
> 2. **Двойное добавление** — два независимых `if` (по `volume >= TRaceVolume` и `volume >= TRaceCylinders`): двигатель, проходящий оба условия, попадёт в `result` **дважды**. К тому же сравнение `volume` (объём) с `TRaceCylinders` (число цилиндров) — почти наверняка логическая ошибка.
> 3. **`car` в `companion object`** — глобальное мутабельное статическое состояние, общее для всех «инстансов» датасорса. Живёт всё время жизни процесса; синглтон-`var` — источник трудноуловимых багов.
> 4. **`saveCar` игнорирует параметр `name`** — всегда пишет `context.getString(R.string.app_name)`. Параметр-обманка.
> 5. **Тип не сходится** — `service.loadAvailableEngines()` по интерфейсу возвращает `ApiEnvelope<LinkedList<Engine>>`, а присваивается в `LinkedList<Engine>` (забыли `.payload`).

> [!tip] Что показывает рефакторинг через корутины
> • **`@Synchronized` (блокирующий монитор) → `Mutex` + `withLock`** — неблокирующая синхронизация, дружелюбная к корутинам (не занимает поток на ожидании).
> • Состояние (`cars`) инкапсулировано в репозитории, без статических `var`.
> • `addCar`/`getLastCar` — `suspend`, доступ к списку только под `withLock` → потокобезопасно при конкурентных корутинах на разных диспетчерах.

> [!note] Чего не хватает «идеальному» решению
> В рефакторинге пропала **фильтрация двигателей** и работа с сетью (`ApiService`) — показан только потокобезопасный CRUD. На собесе стоит проговорить, что фильтрацию выносим в чистую функцию/`UseCase`, а сетевой вызов оборачиваем в `withContext(Dispatchers.IO)` и обрабатываем ошибки (`runCatching`/`Result`), не кидая голый `Exception`.
